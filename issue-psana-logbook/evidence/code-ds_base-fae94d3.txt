  381          """Returns a list of xtc2 files as shown in the db.
  382  
  383          Note that the requested url below returns list of all files (all streams and
  384          all chunks) from the database. For psana2, we only need the first chunk (-c)
  385          of each stream so the original list is filtered down.
  386  
  387          Example of names returned by the requests:
  388          /tmo/tmoc00118/xtc/tmoc00118-r0222-s000-c000.xtc2
  389          /tmo/tmoc00118/xtc/tmoc00118-r0222-s000-c001.xtc2
  390          /tmo/tmoc00118/xtc/tmoc00118-r0222-s002-c000.xtc2
  391  
  392          Returns
  393          file_info['xtc_files']
  394          tmoc00118-r0222-s000-c000.xtc2
  395          tmoc00118-r0222-s002-c000.xtc2
  396          file_info['dirname']
  397          /tmo/tmoc00118/xtc/
  398          """
  399          url = self._file_info_url(runnum)
  400          resp = requests.get(url)
  401          try:
  402              resp.raise_for_status()
  403              all_xtc_files = resp.json()["value"]
  404          except Exception:
  405              self.logger.info(f"Warning: unable to connect to {url}")
  406              all_xtc_files = []
